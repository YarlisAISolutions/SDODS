<TypeTable
  type={{
    name: { type: 'string', description: 'Set automatically from the file name.' },
    description: { type: 'string' },
    'ui.baseUrl': { type: 'url', required: true },
    'api.baseUrl': { type: 'url', required: true },
    'api.headers': { type: 'Record<string, string>', default: '{}' },
    'api.auth': {
      type: 'none | bearer{token} | basic{username,password} | header{name,value} | oauth-client-credentials{tokenUrl,clientId,clientSecret,scope?,audience?}',
      default: '{ type: none }',
      description:
        'Secret fields must be ${VAR} references. oauth-client-credentials posts a client_credentials grant to tokenUrl and sends the access token as a bearer, cached per worker until shortly before expires_in.',
    },
    'api.openapi': {
      type: 'string',
      description: 'Path or URL of the specification for contract steps.',
    },
    aliases: {
      type: 'url[]',
      default: '[]',
      description: 'Other origins of the same environment (recorder rewrites them to routes).',
    },
    db: {
      type: "{ driver: 'sqlite'|'postgres', url }",
      description: 'Test-data database for db datasets.',
    },
    'users.poolSize': { type: 'number', description: 'Maximum accounts a run may lease.' },
    vars: {
      type: 'Record<string, string|number|boolean>',
      default: '{}',
      description: 'Available to steps as {{name}}.',
    },
    use: {
      type: '{ locale?, timezoneId?, geolocation?, permissions?, colorScheme?, ignoreHTTPSErrors?, extraHTTPHeaders?, httpCredentials? }',
      default: '{}',
      description: 'Merged into the browser context options for every run target.',
    },
    screenshots: {
      type: 'partial ScreenshotConfig',
      description: 'Overrides the project section for this environment.',
    },
    evidence: {
      type: '{ trace?, video?, screenshot? }',
      description: 'Overrides single keys of the project `evidence` section for this environment.',
    },
    heal: { type: 'partial HealConfig' },
    timeouts: { type: 'partial Timeouts' },
    'perf.budgets': { type: 'partial PerfBudgets' },
    ci: { type: 'boolean', description: 'Treat this environment as CI (retries.ci applies).' },
  }}

```yaml title="projects/demo-shop/envs/staging.yaml"
name: staging
ui: { baseUrl: https://www.saucedemo.com }
api:
  baseUrl: https://jsonplaceholder.typicode.com
  headers: { Accept: application/json }
  auth: { type: none }
users: { poolSize: 4 }
vars:
  defaultProduct: Sauce Labs Backpack
  standardPassword: '${DEMO_SHOP_PASSWORD:-secret_sauce}'
use: { locale: en-US, timezoneId: America/New_York }
screenshots: { onlyOnFailure: false }
perf: { budgets: { pageLoadMs: 5000 } }
```
